function esc(s){return s.replace(/[&<>"]/g,function(c){return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c];});}
function check(v){
if(!v) return {ok:false,msg:null};
if(!/^[a-zA-Z]/.test(v)) return {ok:false,msg:'Handles start with a letter.'};
if(/[^a-zA-Z0-9_]/.test(v)) return {ok:false,msg:'Letters, numbers and underscore only.'};
if(/__/.test(v)) return {ok:false,msg:'No double underscores.'};
if(v.length < 3) return {ok:false,msg:'A little longer — three characters minimum.'};
return {ok:true,msg:null};
}
function render(){
var v = input.value.trim();
var r = check(v);
var shown = v || 'levi';
plate.innerHTML = '<span class="host">madchatter.chat/</span>' + esc(shown);
if(bigAddr) bigAddr.textContent = shown;
if(plateNm) plateNm.textContent = v ? (shown.charAt(0).toUpperCase() + shown.slice(1)) : 'Levi';
btn.href = v && r.ok ? CLAIM + '?handle=' + encodeURIComponent(v) : CLAIM;
if(!v){ readout.innerHTML = '<span class="hint">' + HINT + '</span>'; return; }
if(!r.ok){ readout.innerHTML = '<span class="warn">' + esc(r.msg) + '</span>'; return; }
// Deliberately NOT an availability claim — only the server knows that.
readout.innerHTML =
'<span class="addr">madchatter.chat/<span class="tail">' + esc(v) + '</span></span>' +
'<span class="status"><i class="dot"></i>Your live address</span>';
}

input.addEventListener('keydown', function(e){ if(e.key === 'Enter') btn.click(); });

box.hidden = false;
}).catch(function(){});
}
